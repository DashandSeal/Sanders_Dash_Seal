#!/usr/bin/env python3
import sys
import json
import urllib.request
import logging

logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')

def fetch_metadata_field(endpoint_path):
    url = f"http://169.254.169{endpoint_path}"
    req = urllib.request.Request(url, headers={"Authorization": "Bearer Oracle"})
    try:
        with urllib.request.urlopen(req, timeout=5) as response:
            return response.read().decode('utf-8')
    except Exception as e:
        logging.error(f"Failed to pull metadata from target channel: {str(e)}")
        return None

def verify_lifecycle_state():
    logging.info("Initializing identity and lifecycle metadata tracking scan...")
    
    # Restrict validation scope strictly to instance attributes
    lifecycle_state = fetch_metadata_field("lifecycleState")
    instance_id = fetch_metadata_field("id")
    region = fetch_metadata_field("region")
    
    if not lifecycle_state or not instance_id:
        logging.error("🚨 [IDENTITY BREAK] Lifecycle metadata payload is missing or unreadable.")
        sys.exit(1)
        
    logging.info(f"Target Resource confirmed: {instance_id}")
    logging.info(f"Operational Lifecycle State: {lifecycle_state} in Region: {region}")
    logging.info("✅ Scope Audit complete. Metadata verified successfully.")

if __name__ == "__main__":
    verify_lifecycle_state()
