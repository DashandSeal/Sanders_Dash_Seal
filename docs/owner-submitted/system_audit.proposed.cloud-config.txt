write_files:
  - path: /opt/sanders-dash-seal/scripts/system_preflight_audit.sh
    owner: root:root
    permissions: '0755'
    content: |
      #!/bin/bash
      set -Eeuo pipefail
      
      LOG_FILE="/var/log/sds_platform_preflight.log"
      {
        echo "======================================================================"
        echo "AUTOMATED TELEMETRY INVENTORY RUNTIME ENGINE"
        echo "Timestamp: \$(date -u '+%Y-%m-%d %H:%M:%S UTC')"
        echo "======================================================================"
        
        echo "--- [1/4] SYSTEM ARTIFACT ARCHITECTURE ---"
        uname -m
        
        echo -e "\n--- [2/4] SYSTEM BINARY LOCATIONS ---"
        for binary in git docker oci; do
            command -v "\$binary" || echo "\$binary: NOT FOUND"
        done
        
        echo -e "\n--- [3/4] CONTAINER ORCHESTRATION LAYER ---"
        if command -v docker &>/dev/null; then
            docker compose version
        fi
        
        echo -e "\n--- [4/4] ACL PROPERTY CEILINGS ---"
        PROTECTED_PATHS=("/opt/sanders-dash-seal" "/opt/sds-approved-source" "/etc/sanders-dash-seal")
        for target_path in "\${PROTECTED_PATHS[@]}"; do
            if [ -d "\$target_path" ]; then
                ls -ld "\$target_path"
            else
                echo "Path tracking missing: \$target_path"
            fi
        done
      } > "$LOG_FILE" 2>&1

runcmd:
  - echo "Initializing background environment configuration loop..."
  # 1. Force execute the diagnostic verification blueprint instantly
  - /opt/sanders-dash-seal/scripts/system_preflight_audit.sh
  
  # 2. Register the audit tracking results directly into standard syslog files
  - cat /var/log/sds_platform_preflight.log
