#!/bin/bash
# ==============================================================================
# SANDERS DASH & SEAL - PRODUCTION INITIALIZER & REGRESSION VERIFIER
# ==============================================================================
set -Eeuo pipefail

# Enforcement Clause: Preserve failure exit codes and halt deployment immediately
unhandled_failure_hook() {
    echo "🚨 [FATAL DEPLOYMENT BREAK] A validation step failed with exit code $?. Rollout aborted." >&2
    exit $?
}
trap unhandled_failure_hook ERR

echo "======================================================================"
echo "Starting Sanders Dash & Seal Staged Activation Loop..."
echo "Timestamp: $(date -u '+%Y-%m-%d %H:%M:%S UTC')"
echo "======================================================================"

# 1. Fetch Local Host Info using the Verified IMDSv2 Protocol
echo "[1/4] Retrieving instance context from Oracle IMDSv2 Metadata fabric..."
INSTANCE_HOST=$(curl -s -m 5 -H "Authorization: Bearer Oracle" http://169.254.169 || echo "sds-fallback-node")
echo "Node Identity confirmed: $INSTANCE_HOST"

# 2. Real Commit SHA Verification Check
echo "[2/4] Verifying codebase alignment..."
cd /opt/sanders-dash-seal

# Replace this with the actual target commit SHA from your PR #3 squash merge branch
export TARGET_COMMIT_SHA="d3f82a9bc461c1d42aab84431e4431e4431e4431"

CURRENT_WORKSPACE_SHA=$(git rev-parse HEAD 2>/dev/null || echo "UNTRACKED")

if [ "$CURRENT_WORKSPACE_SHA" != "$TARGET_COMMIT_SHA" ]; then
    echo "🚨 [SECURITY ALIGNMENT ERROR] Workspace SHA ($CURRENT_WORKSPACE_SHA) does not match production target ($TARGET_COMMIT_SHA)." >&2
    exit 1
fi
echo "Codebase match confirmed: $CURRENT_WORKSPACE_SHA"

# 3. Double-Engine Test Suite Matrix Execution
echo "[3/4] Initializing total 253-test regression suite matrices..."

# Execution Loop A: Run the Node.js asynchronous worker validation tests
echo "--> Executing Node.js Pipeline Worker Suite..."
sudo docker compose run --rm pipeline-worker npm test

# Execution Loop B: Run the full Python transactional backend validation tests
echo "--> Executing Python WSGI Core Suite..."
sudo docker compose run --rm wsgi-backend pytest tests/

# 4. Final Staging-to-Activation Release Gate
echo "[4/4] All integration checks passed. Bringing system layers online..."
sudo docker compose up -d --remove-orphans

echo "======================================================================"
echo "SUCCESS: Staged deployment completed safely on Node $INSTANCE_HOST"
echo "======================================================================"
