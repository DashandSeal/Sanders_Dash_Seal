name: "Sanders Dash & Seal - Platinum Forensic Security Audit"

on:
  push:
    branches: [ "main", "release/*" ]
  pull_request:
    branches: [ "main" ]
  schedule:
    - cron: '0 */6 * * *' # Automated health audit scans every 6 hours

jobs:
  forensic-integrity-checksum:
    runs-on: ubuntu-latest
    steps:
      - name: "1. Checkout Secure Code Base"
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: "2. Generate Cryptographic Manifest Digests"
        id: generate_hashes
        run: |
          echo "========================================================================"
          echo "Initializing SHA-256 Code Base Verification Loops..."
          echo "========================================================================"
          
          # Create an inventory manifest excluding git directories and tracking outputs
          find . -type f -not -path '*/.*' -not -path './forensic_manifest.sha256' | sort > file_list.txt
          
          # Calculate individual SHA-256 shasums for every source file
          while IFS= read -r file; do
            sha256sum "$file" >> forensic_manifest.sha256
          done < file_list.txt
          
          echo "Cryptographic baseline manifest constructed completely."
          cat forensic_manifest.sha256

      - name: "3. Evaluate Execution Controls & Drift Anomalies"
        id: audit_drift
        run: |
          # Audit for executable permission tampering on configurations
          if [ "$(stat -c "%a" ./worker/Dockerfile)" != "644" ] && [ "$(stat -c "%a" ./worker/Dockerfile)" != "default" ]; then
            echo "TAMPER_ALERT=true" >> $GITHUB_ENV
            echo "REASON=Dockerfile permissions anomaly detected." >> $GITHUB_ENV
          fi

          # Detect code duplication indicators or unexpected additions
          git status --porcelain > drift.log
          if [ -s drift.log ]; then
            echo "TAMPER_ALERT=true" >> $GITHUB_ENV
            echo "REASON=Untracked tracking or uncommitted script structural drift detected." >> $GITHUB_ENV
          fi

      - name: "4. Trigger Incident Response System"
        if: env.TAMPER_ALERT == 'true' || failure()
        uses: dawidd6/action-send-mail@v3
        with:
          server_address: ${{ secrets.SDS_SMTP_SERVER }}
          server_port: 465
          username: ${{ secrets.SDS_SMTP_USERNAME }}
          password: ${{ secrets.SDS_SMTP_PASSWORD }}
          subject: "🚨 [CRITICAL ALERT] Sanders Dash & Seal Code Base Tampering Violation"
          to: "rooseveltsanders381@gmail.com"
          from: "Forensic Integrity Agent <security@sandersdashseal.app>"
          secure: true
          body: |
            ======================================================================
            SANDERS DASH & SEAL - AUTOMATED FORENSIC INCIDENT RESPONSE LOG
            ======================================================================
            Timestamp: ${{ github.event.repository.updated_at }} UTC
            Triggering Event: Codebase Modification / Alteration Scan Failure
            Actor/Commiter: ${{ github.actor }}
            Commit Hash: ${{ github.sha }}
            
            DIAGNOSTICS & BREACH VECTOR ANALYSIS:
            - Alert Status: PLATINUM ACTION REQUIRED
            - Violation Indicators: Unauthorized code duplication or layout anomaly detected.
            - Trace Log Ingress: Check GitHub Actions Run ID: ${{ github.run_id }}
            
            This notification was systematically generated by the automated forensic tamper-response pipeline. Immediate cryptographic audit review is mandated.
