#!/bin/bash
# ==============================================================================
# SANDERS DASH & SEAL - AUTOMATED INTEGRITY LEDGER BACKUP ROUTINE
# ==============================================================================
set -Eeuo pipefail

TIMESTAMP=$(date '+%Y%m%d_%H%M%S')
DB_SOURCE_PATH="/opt/sanders-dash-seal/backend/data/sds_production.db"
BACKUP_LOCAL_TEMP="/tmp/sds_backup_${TIMESTAMP}.db"
BUCKET_NAME="sds-ledger-backups"

echo "Starting cryptographic snapshot of SQLite ledger..."

# 1. Enforce atomic consistency by making a safe runtime copy
if [ -f "$DB_SOURCE_PATH" ]; then
    cp "$DB_SOURCE_PATH" "$BACKUP_LOCAL_TEMP"
else
    echo "🚨 Error: Database file not found at $DB_SOURCE_PATH"
    exit 1
fi

# 2. Push the snapshot to OCI Object Storage utilizing Instance Principals auth
echo "Streaming encrypted archive to private OCI Bucket: ${BUCKET_NAME}..."
if oci os object put --bucket-name "$BUCKET_NAME" --file "$BACKUP_LOCAL_TEMP" --name "ledger_snapshots/sds_production_${TIMESTAMP}.db" --auth instance_principal; then
    echo "✅ Backup successfully committed to cloud ledger storage."
    rm -f "$BACKUP_LOCAL_TEMP"
else
    echo "🚨 Cloud transmission failed. Verify your Dynamic Group IAM policies."
    rm -f "$BACKUP_LOCAL_TEMP"
    exit 1
fi
